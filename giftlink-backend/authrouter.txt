Register
curl --location 'http://localhost:3060/api/auth/register' \
--header 'Content-Type: application/json' \
--data-raw '{
    "firstName": "Test",
    "lastName": "User",
    "email": "testuser123@example.com",
    "password": "test123"
}'




LOGIN

curl --location 'http://localhost:3060/api/auth/login' \
--header 'Content-Type: application/json' \
--data-raw '{
    "email": "doesnotexist@example.com",
    "password": "test123"
}'





curl --location --request PUT 'http://localhost:3060/api/auth/update' \
--header 'Content-Type: application/json' \
--header 'email: testuser123@example.com' \
--data-raw '{
    "name": "UpdatedName"
}'
